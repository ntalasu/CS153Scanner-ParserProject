PROGRAM SimpleCaseTest;

BEGIN
    i := 3;

    CASE i+1 OF
        1:       j := i;
        -8:      j := 8*i;
        5, 7, 4: j := 574*i;
    END;

    write('j = ');
    writeln(j);

    i := -4;

    CASE i OF
        -4, -2, 0, 2, 4 : k := 100;
        1, 3, 5         : k := 200;
    END;

    write('k = ');
    writeln(k)
END.
